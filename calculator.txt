<script>
  const displayEl = document.getElementById('display');
  const historyEl = document.getElementById('history');
  const previewEl = document.getElementById('preview');
  const ledEl = document.getElementById('led');

  const state = {
    current: '0',
    previous: null,
    operator: null,
    expression: '',
    resetNext: false,
    errored: false
  };

  function compute(a, b, op){
    switch(op){
      case '+': return a + b;
      case '−': return a - b;
      case '×': return a * b;
      case '÷': return b === 0 ? NaN : a / b;
      default: return b;
    }
  }

  function trimNumber(num){
    if(!isFinite(num)) return 'Error';
    const rounded = parseFloat(num.toPrecision(12));
    return rounded.toString();
  }

  function formatForDisplay(str){
    if(str === 'Error') return str;
    const negative = str.startsWith('-');
    const raw = negative ? str.slice(1) : str;
    let [intPart, decPart] = raw.split('.');
    intPart = intPart === '' ? '0' : intPart;
    const withCommas = intPart.replace(/\B(?=(\d{3})+(?!\d))/g, ',');
    const result = withCommas + (decPart !== undefined ? '.' + decPart : '');
    return (negative ? '-' : '') + result;
  }

  function updateView(){
    displayEl.textContent = state.errored ? 'Error' : formatForDisplay(state.current);
    historyEl.innerHTML = state.expression || '&nbsp;';

    if(state.operator && state.previous !== null && !state.resetNext && !state.errored){
      const b = parseFloat(state.current);
      if(!isNaN(b)){
        const result = compute(state.previous, b, state.operator);
        previewEl.textContent = '= ' + (isFinite(result) ? formatForDisplay(trimNumber(result)) : 'Error');
      } else {
        previewEl.innerHTML = '&nbsp;';
      }
    } else {
      previewEl.innerHTML = '&nbsp;';
    }

    ledEl.classList.toggle('led-err', state.errored);
  }

  function pulseDisplay(){
    displayEl.classList.add('pulse');
    setTimeout(() => displayEl.classList.remove('pulse'), 180);
  }

  function triggerError(){
    state.current = 'Error';
    state.previous = null;
    state.operator = null;
    state.expression = '';
    state.resetNext = true;
    state.errored = true;
    updateView();
  }

  function inputDigit(d){
    if(state.errored) clearAll();

    if(state.resetNext){
      state.current = d === '.' ? '0.' : d;
      state.resetNext = false;
    } else if(d === '.'){
      if(!state.current.includes('.')) state.current += '.';
    } else {
      state.current = state.current === '0' ? d : state.current + d;
    }

    if(state.current.replace('-', '').replace('.', '').length > 15) return;
    updateView();
  }

  function setOperator(op){
    if(state.errored) return;
    const cur = parseFloat(state.current);

    if(state.operator && !state.resetNext){
      const result = compute(state.previous, cur, state.operator);
      if(!isFinite(result)){ triggerError(); return; }
      state.previous = parseFloat(trimNumber(result));
      state.expression = formatForDisplay(trimNumber(state.previous)) + ' ' + op;
    } else {
      state.previous = cur;
      state.expression = formatForDisplay(trimNumber(cur)) + ' ' + op;
    }

    state.operator = op;
    state.resetNext = true;
    updateView();
  }

  function equals(){
    if(state.errored || state.operator === null || state.previous === null) return;
    const cur = parseFloat(state.current);
    const result = compute(state.previous, cur, state.operator);
    if(!isFinite(result)){ triggerError(); return; }

    state.expression = formatForDisplay(trimNumber(state.previous)) + ' ' + state.operator + ' ' + formatForDisplay(trimNumber(cur)) + ' =';
    state.current = trimNumber(result);
    state.previous = null;
    state.operator = null;
    state.resetNext = true;
    updateView();
    pulseDisplay();
  }

  function clearAll(){
    state.current = '0';
    state.previous = null;
    state.operator = null;
    state.expression = '';
    state.resetNext = false;
    state.errored = false;
    updateView();
  }

  function deleteLast(){
    if(state.errored){ clearAll(); return; }
    if(state.resetNext) return;
    state.current = state.current.length > 1 ? state.current.slice(0, -1) : '0';
    updateView();
  }

  function percent(){
    if(state.errored) return;
    const val = parseFloat(state.current);
    if(isNaN(val)) return;
    state.current = trimNumber(val / 100);
    state.resetNext = false;
    updateView();
  }

  document.querySelectorAll('.key').forEach(btn => {
    btn.addEventListener('click', () => {
      if(btn.dataset.digit !== undefined) inputDigit(btn.dataset.digit);
      else if(btn.dataset.action === 'decimal') inputDigit('.');
      else if(btn.dataset.action === 'operator') setOperator(btn.dataset.op);
      else if(btn.dataset.action === 'equals') equals();
      else if(btn.dataset.action === 'clear') clearAll();
      else if(btn.dataset.action === 'delete') deleteLast();
      else if(btn.dataset.action === 'percent') percent();
    });
  });

  function flashKey(selector){
    const btn = document.querySelector(selector);
    if(!btn) return;
    btn.classList.add('pressed');
    setTimeout(() => btn.classList.remove('pressed'), 120);
  }

  document.addEventListener('keydown', (e) => {
    const key = e.key;
    if(/^[0-9]$/.test(key)){ inputDigit(key); flashKey(`[data-digit="${key}"]`); }
    else if(key === '.'){ inputDigit('.'); flashKey('[data-action="decimal"]'); }
    else if(key === '+'){ setOperator('+'); flashKey('[data-op="+"]'); }
    else if(key === '-'){ setOperator('−'); flashKey('[data-op="−"]'); }
    else if(key === '*'){ setOperator('×'); flashKey('[data-op="×"]'); }
    else if(key === '/'){ e.preventDefault(); setOperator('÷'); flashKey('[data-op="÷"]'); }
    else if(key === 'Enter' || key === '='){ e.preventDefault(); equals(); flashKey('[data-action="equals"]'); }
    else if(key === 'Backspace'){ deleteLast(); flashKey('[data-action="delete"]'); }
    else if(key === 'Escape'){ clearAll(); flashKey('[data-action="clear"]'); }
    else if(key === '%'){ percent(); flashKey('[data-action="percent"]'); }
  });

  updateView();
</script>
